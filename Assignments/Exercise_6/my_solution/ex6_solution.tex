\subsection*{Exercise 1: Information-Theory Basics}

\begin{enumerate}[label=(\alph*)]
    \item {
        \textbf{Answer: } Cross-Entropy: \(H(p,q) = - \Sigma_{x \in X} p(x) \log q(x)\) is to be minimized, equivalent to minimizing KL Divergence: \(D_{KL}(p||q) = \Sigma_{x \in X} p(x) \log \frac{p(x)}{q(x)}\).
    }
    \item {
        \textbf{Answer: } By maximizing log-likelihood, \(log q_\theta(w)\), will minimize the cross-entropy since \(H(p,q_\theta) = -\mathbb{E}_p[\log q_\theta(w)] \propto -\log q_\theta(w)\).
    }
    \item \textbf{Answer: } Proof of \(H(p,q_\theta) = H(p) + D_{KL}(p||q)\): {
        \begin{align*}
            H(p,q_\theta) &= -\mathbb{E}_p \log q(x) \\
                          &= \mathbb{E}_p (\log p(x) - \log p(x) - \log q_\theta(x)) \\
                          &= -\mathbb{E}_p \log p(x) + \mathbb{E}_p \frac{\log p(x)}{\log q_\theta(x)} \\
                          &= H(p) + D_{KL}(p||q)
        \end{align*}
    }
    \item 
\end{enumerate}
